C2 (5 marks) TRICKY
This Hero forgot to call super().__init__ — so name and hp never get set. Add the missing line at the start of __init__.
class Character:
    def __init__(self, name, hp):
        self.name = name
        self.hp = hp

class Hero(Character):
    def __init__(self, name, hp, weapon):
        # ← MISSING: super().__init__(name, hp)
        self.weapon = weapon

h = Hero("Arrow", 100, "Sword")
print(h.name)   # AttributeError: 'Hero' object has no attribute 'name'
